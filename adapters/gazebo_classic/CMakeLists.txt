cmake_minimum_required(VERSION 3.16)
project(msim_gazebo_sensors LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(gazebo REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs nav_msgs std_msgs livox_ros_driver)
set(PX4_GAZEBO_BUILD "" CACHE PATH "PX4 Gazebo Classic build directory with SITLGps protobuf")
include_directories(${GAZEBO_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS} ${PX4_GAZEBO_BUILD})
link_directories(${GAZEBO_LIBRARY_DIRS} ${PX4_GAZEBO_BUILD})
add_library(msim_gps_relay SHARED gps_relay.cpp)
add_library(msim_scanning_lidar SHARED scanning_lidar.cpp)
foreach(target msim_gps_relay msim_scanning_lidar)
  target_compile_options(${target} PRIVATE -O2 -Wno-deprecated-declarations)
  target_link_libraries(${target} ${GAZEBO_LIBRARIES} ${catkin_LIBRARIES})
endforeach()
target_link_libraries(msim_gps_relay "${PX4_GAZEBO_BUILD}/libsensor_msgs.so")
