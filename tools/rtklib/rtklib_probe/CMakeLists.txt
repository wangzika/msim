cmake_minimum_required(VERSION 3.5)
project(msim_rtklib_probe)
find_package(catkin REQUIRED COMPONENTS rtklib)
find_package(Eigen3 REQUIRED)
catkin_package()
add_executable(msim_rtklib_probe main.cpp)
target_include_directories(msim_rtklib_probe PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(msim_rtklib_probe Eigen3::Eigen)
target_link_libraries(msim_rtklib_probe ${catkin_LIBRARIES})
